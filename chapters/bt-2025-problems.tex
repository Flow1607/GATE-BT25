\item A thermometer measuring body temperature follows a first-order response with a time constant of $40$ seconds. The instrument will reach $95\%$ of its steady-state output at \underline{\hspace{1.5cm}} seconds (round off to the nearest integer).
\hfill (BT~2025)
\begin{enumerate}[itemsep=0pt]
    \begin{multicols}{2}
        \item $60$
        \item $80$
        \item $120$
        \item $160$
    \end{multicols}
\end{enumerate}

\solution
The dynamic thermal response of the thermometer is modeled by the standard first-order differential equation given in \eqref{eq:EE26BTECH11213-first_order_ode_std}:
\begin{align}
\tau \frac{dy}{dt} + y\brak{t} = y_{ss}
\end{align}
which is analogous to the series $RC$ low-pass circuit in \eqref{eq:EE26BTECH11213-rc_circuit_ode}, where the measured temperature $y\brak{t}$ maps to the capacitor output voltage $V_{out}\brak{t}$, the steady-state reading $y_{ss}$ corresponds to $V_{dc}$, and the thermal time constant $\tau = 40\text{ s}$ corresponds to the circuit time constant $RC$.

Converting the system to the $s$-domain equivalent circuit from \eqref{eq:EE26BTECH11213-rc_s_domain_vout}:
\begin{align}
Y\brak{s} = \frac{y_{ss}}{s\brak{\tau s + 1}} = y_{ss}\sbrak{\frac{1}{s} - \frac{1}{s + \frac{1}{\tau}}}
\end{align}
Taking the inverse Laplace transform directly yields the transient response from \eqref{eq:EE26BTECH11213-first_order_step_sol}:
\begin{align}
y\brak{t} = y_{ss}\brak{1 - e^{-\frac{t}{\tau}}}
\end{align}

The instrument reaches $95\%$ of its steady-state reading when $y\brak{t_{95}} = 0.95 y_{ss}$:
\begin{align}
y_{ss}\brak{1 - e^{-\frac{t_{95}}{\tau}}} &= 0.95 y_{ss} \\
1 - e^{-\frac{t_{95}}{\tau}} &= 0.95 \\
e^{-\frac{t_{95}}{\tau}} &= 0.05 = \frac{1}{20}
\end{align}

Taking the natural logarithm on both sides:
\begin{align}
-\frac{t_{95}}{\tau} &= \ln\brak{0.05} = -\ln\brak{20} \\
t_{95} &= \tau \ln\brak{20}
\end{align}
Substituting the given time constant $\tau = 40\text{ s}$:
\begin{align}
t_{95} &= 40 \ln\brak{20} \\

\end{align}

\end{align}

Using \eqref{eq:EE26BTECH11213-linear_identity} and \eqref{eq:EE26BTECH11213-linear_coeffs_zero}, since this equation must hold for any constants $C_1$ and $C_2$:
\begin{align}
1 + m + n &= 0 \label{eq:EE26BTECH11213-q45_eq1} \\
1 - m + n &= 0 \label{eq:EE26BTECH11213-q45_eq2}
\end{align}

Adding \eqref{eq:EE26BTECH11213-q45_eq1} and \eqref{eq:EE26BTECH11213-q45_eq2}:
\begin{align}
2 + 2n = 0 \implies n = -1
\end{align}

Subtracting \eqref{eq:EE26BTECH11213-q45_eq2} from \eqref{eq:EE26BTECH11213-q45_eq1}:
\begin{align}
2m = 0 \implies m = 0
\end{align}

Evaluating $m + n$:
\begin{align}
m + n = 0 + \brak{-1} = -1
\end{align}

Hence, option (B) is correct.

\item Consider the piecewise function
\begin{align}
f\brak{x} = \begin{cases}
\sin\brak{2x}, & x > 0 \\
a + bx, & x \le 0
\end{cases}
\end{align}
where $a$ and $b$ are real constants. If $f\brak{x}$ is differentiable at $x = 0$, then the value of $a + b$ is
\hfill (BT~2025)
\begin{enumerate}[itemsep=0pt]
    \begin{multicols}{2}
        \item 0
        \item 1
        \item 2
        \item 3
    \end{multicols}
\end{enumerate}

\solution
Since $f\brak{x}$ is differentiable at $x = 0$, from \eqref{eq:EE26BTECH11213-diff_implies_cont} it must also be continuous at $x = 0$. Applying the continuity condition from \eqref{eq:EE26BTECH11213-continuity_cond}:
\begin{align}
\lim_{x \to 0^+} f\brak{x} = \lim_{x \to 0^-} f\brak{x} = f\brak{0}
\end{align}

Evaluating the one-sided limits:
\begin{align}
\lim_{x \to 0^+} f\brak{x} &= \lim_{x \to 0^+} \sin\brak{2x} = \sin\brak{0} = 0 \\
\lim_{x \to 0^-} f\brak{x} &= \lim_{x \to 0^-} \brak{a + bx} = a + b\brak{0} = a
\end{align}
Equating the limits yields:
\begin{align}
a = 0
\end{align}

Applying the differentiability condition and matching one-sided derivatives at $x = 0$ using \eqref{eq:EE26BTECH11213-slope_matching}:
\begin{align}
f'\brak{0^+} = f'\brak{0^-}
\end{align}

For $x > 0$:
\begin{align}
f'\brak{x} = \frac{d}{dx}\brak{\sin\brak{2x}} = 2\cos\brak{2x} \implies f'\brak{0^+} = 2\cos\brak{0} = 2
\end{align}

For $x < 0$:
\begin{align}
f'\brak{x} = \frac{d}{dx}\brak{a + bx} = b \implies f'\brak{0^-} = b
\end{align}

Equating the two slopes:
\begin{align}
b = 2
\end{align}

Evaluating $a + b$:
\begin{align}
a + b = 0 + 2 = 2
\end{align}
Hence, option (C) is correct.

\item In a fermentation process, each mole of glucose is converted to biomass $\brak{\text{CH}_{1.8}\text{O}_{0.5}\text{N}_{0.2}}$, with a biomass yield coefficient of $0.4\text{ C-mol/C-mol}$, according to the unbalanced equation given below.
\begin{align}
\text{C}_6\text{H}_{12}\text{O}_6 + \text{NH}_3 + \text{O}_2 \rightarrow \text{CH}_{1.8}\text{O}_{0.5}\text{N}_{0.2} + \text{CO}_2 + \text{H}_2\text{O}

\item Let $a_0 = 0$ and define $a_n = \frac{1}{2}\brak{1 + a_{n-1}}$ for all positive integers $n \ge 1$.
The least value of $n$ for which $\abs{1 - a_n} < \frac{1}{2^{10}}$ is \underline{\hspace{2cm}}. 
\hfill (BT~2025)

\solution
The given recurrence relation is:
\begin{align}
a_n = \frac{1}{2}\brak{1 + a_{n-1}}, \quad n \ge 1, \quad a_0 = 0
\end{align}

Extending the difference equation across all $n \in \mathbb{Z}$ by causal multiplication with $u\sbrak{n}$ from \eqref{eq:EE26BTECH11213-discrete_unit_step}:
\begin{align}
a_n u\sbrak{n} - \frac{1}{2} a_{n-1} u\sbrak{n-1} = \frac{1}{2} u\sbrak{n-1}
\end{align}

Let $A\brak{z} = \mathcal{Z}\sbrak{a_n u\sbrak{n}}$. Applying the time-shift property \eqref{eq:EE26BTECH11213-z_shift_property} and the delayed step transform \eqref{eq:EE26BTECH11213-z_delayed_step}:
\begin{align}
A\brak{z} - \frac{1}{2} z^{-1} A\brak{z} &= \frac{1}{2} \mathcal{Z}\sbrak{u\sbrak{n-1}} \\
A\brak{z}\brak{1 - \frac{1}{2} z^{-1}} &= \frac{\frac{1}{2} z^{-1}}{1 - z^{-1}}
\end{align}

Solving algebraically for $A\brak{z}$:
\begin{align}
A\brak{z} = \frac{\frac{1}{2} z^{-1}}{\brak{1 - z^{-1}}\brak{1 - \frac{1}{2} z^{-1}}}, \quad \abs{z} > 1
\end{align}

Performing partial fraction decomposition with $w = z^{-1}$:
\begin{align}
\frac{\frac{1}{2} w}{\brak{1 - w}\brak{1 - \frac{1}{2} w}} = \frac{C_1}{1 - w} + \frac{C_2}{1 - \frac{1}{2} w}
\end{align}
where
\begin{align}
C_1 &=  \frac{\frac{1}{2} w}{1 - \frac{1}{2} w} _{w = 1} = \frac{1/2}{1/2} = 1 \\
C_2 &=  \frac{\frac{1}{2} w}{1 - w} _{w = 2} = \frac{1}{1 - 2} = -1
\end{align}

Substituting back into $A\brak{z}$:
\begin{align}
A\brak{z} = \frac{1}{1 - z^{-1}} - \frac{1}{1 - \frac{1}{2} z^{-1}}
\end{align}

Applying the inverse Z-transform using \eqref{eq:EE26BTECH11213-z_step_transform} and \eqref{eq:EE26BTECH11213-z_inv_exponential}:
\begin{align}
a_n = \sbrak{1 - \brak{\frac{1}{2}}^n} u\sbrak{n} = 1 - 2^{-n}, \quad n \ge 0
\end{align}

Applying the stopping threshold condition $\abs{1 - a_n} < \frac{1}{2^{10}}$:
\begin{align}
\abs{1 - \brak{1 - 2^{-n}}} &< \frac{1}{2^{10}} \\
2^{-n} &< 2^{-10} \\
2^n &> 2^{10} \implies n > 10
\end{align}

Since $n$ must be an integer, the minimal integer satisfying the strict inequality is:
\begin{align}
n = 11
\end{align}
