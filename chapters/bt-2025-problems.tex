\item A circle with center at $\brak{x, y} = \brak{0.5, 0}$ and radius $r_1 = 0.5$ intersects with another circle with center at $\brak{x, y} = \brak{1, 1}$ and radius $r_2 = 1$ at two points. One of the points of intersection $\brak{x, y}$ is:
\hfill (GATE~BT~2025)
\begin{enumerate}[itemsep=0pt]
    \begin{multicols}{2}
        \item $\brak{0, 0}$
        \item $\brak{0.2, 0.4}$
        \item $\brak{0.5, 0.5}$
        \item $\brak{1, 2}$
    \end{multicols}
\end{enumerate}

\solution
Let $\vec{x} = \myvec{x \\ y}$. The general second-degree equation of a conic in vector form is:
\begin{align}
\vec{x}^\top \vec{V} \vec{x} + 2\vec{u}^\top \vec{x} + f = 0 \label{eq:conic_gen_form}
\end{align}
For a circle with center vector $\vec{c}$ and radius $r$, the geometric locus satisfies:
\begin{align}
\norm{\vec{x} - \vec{c}}^2 = r^2
\end{align}
Expanding the norm in vector product form:
\begin{align}
\brak{\vec{x} - \vec{c}}^\top \brak{\vec{x} - \vec{c}} &= r^2 \\
\vec{x}^\top \vec{x} - 2\vec{c}^\top \vec{x} + \vec{c}^\top \vec{c} - r^2 &= 0
\end{align}
Comparing with \eqref{eq:conic_gen_form}, the conic matrix parameters are:
\begin{align}
\vec{V} = \vec{I} = \myvec{1 & 0 \\ 0 & 1}, \quad \vec{u} = -\vec{c}, \quad f = \norm{\vec{c}}^2 - r^2 = \vec{c}^\top \vec{c} - r^2
\end{align}

For the two given circles:
\begin{align}
\text{Circle 1: } \vec{c}_1 = \myvec{0.5 \\ 0}, \; r_1 = 0.5 &\implies \vec{x}^\top \vec{x} - 2\myvec{0.5 & 0}\vec{x} + \brak{0.5^2 - 0.5^2} = 0 \\
\vec{V}_1 = \myvec{1 & 0 \\ 0 & 1}, \quad \vec{u}_1 &= -\vec{c}_1 = \myvec{-0.5 \\ 0}, \quad f_1 = \norm{\vec{c}_1}^2 - r_1^2 = 0 \\
\text{Circle 2: } \vec{c}_2 = \myvec{1 \\ 1}, \; r_2 = 1 &\implies \vec{x}^\top \vec{x} - 2\myvec{1 & 1}\vec{x} + \brak{1^2 + 1^2 - 1^2} = 0 \\
\vec{V}_2 = \myvec{1 & 0 \\ 0 & 1}, \quad \vec{u}_2 &= -\vec{c}_2 = \myvec{-1 \\ -1}, \quad f_2 = \norm{\vec{c}_2}^2 - r_2^2 = 1
\end{align}

Using the pencil of conics from \eqref{eq:pencil_conics}, the condition for a degenerate conic from \eqref{eq:pencil_det_zero} gives:
\begin{align}
\mydet{\vec{V}_1 + \mu \vec{V}_2 & \vec{u}_1 + \mu \vec{u}_2 \\ \brak{\vec{u}_1 + \mu \vec{u}_2}^\top & f_1 + \mu f_2} = 0
\end{align}
Substituting the parameter matrices:
\begin{align}
\mydet{1 + \mu & 0 & -0.5 - \mu \\
0 & 1 + \mu & -\mu \\
-0.5 - \mu & -\mu & \mu} = 0
\end{align}
Expanding the determinant:
\begin{align}
\brak{1 + \mu}\sbrak{\brak{1 + \mu}\mu - \mu^2} - \brak{-0.5 - \mu}\sbrak{0 - \brak{1 + \mu}\brak{-0.5 - \mu}} &= 0 \\
\brak{1 + \mu}\mu - \brak{1 + \mu}\brak{\mu + 0.5}^2 &= 0 \\
\brak{1 + \mu}\brak{-\mu^2 - \frac{1}{4}} &= 0
\end{align}
Since $\mu^2 + \frac{1}{4} = 0$ yields imaginary roots, the only real root is:
\begin{align}
\mu = -1
\end{align}

Substituting $\mu = -1$ into \eqref{eq:pencil_conics} eliminates the second-degree terms $\brak{\vec{V}_1 - \vec{V}_2 = \vec{0}}$, yielding the common chord (radical axis):
\begin{align}
2\brak{\vec{u}_1 - \vec{u}_2}^\top \vec{x} + \brak{f_1 - f_2} &= 0 \\
2\brak{\myvec{-0.5 \\ 0} - \myvec{-1 \\ -1}}^\top \vec{x} + \brak{0 - 1} &= 0 \\
2\myvec{0.5 & 1}\vec{x} - 1 = 0 \implies \myvec{1 & 2}\vec{x} &= 1
\end{align}
Thus, the equation of the line passing through the intersection points is $x + 2y = 1$.

The normal vector of this line is $\vec{n} = \myvec{1 \\ 2}$. A direction vector $\vec{m}$ orthogonal to $\vec{n}$ ($\vec{m}^\top \vec{n} = 0$) and a point $\vec{h}$ on the line are:
\begin{align}
\vec{m} = \myvec{2 \\ -1}, \quad \vec{h} = \myvec{1 \\ 0}
\end{align}
Expressing the line in parametric form $\vec{x} = \vec{h} + \kappa \vec{m}$ and substituting into Circle 1 using \eqref{eq:conic_kappa_roots}:
\begin{align}
\vec{m}^\top \vec{V}_1 \vec{m} &= \myvec{2 & -1}\myvec{1 & 0 \\ 0 & 1}\myvec{2 \\ -1} = 5 \\
\vec{V}_1 \vec{h} + \vec{u}_1 &= \myvec{1 \\ 0} + \myvec{-0.5 \\ 0} = \myvec{0.5 \\ 0} \\
\vec{m}^\top \brak{\vec{V}_1 \vec{h} + \vec{u}_1} &= \myvec{2 & -1}\myvec{0.5 \\ 0} = 1 \\
g\brak{\vec{h}} &= \vec{h}^\top \vec{V}_1 \vec{h} + 2\vec{u}_1^\top \vec{h} + f_1 = 1 - 1 + 0 = 0
\end{align}
Evaluating the parameter roots $\kappa$:
\begin{align}
\kappa = \frac{-1 \pm \sqrt{1^2 - 0}}{5} = \frac{-1 \pm 1}{5} \implies \kappa_1 = 0, \quad \kappa_2 = -\frac{2}{5}
\end{align}

The points of intersection are:

\vec{x}_2 &= \vec{h} - \frac{2}{5}\vec{m} = \myvec{1 \\ 0} - \frac{2}{5}\myvec{2 \\ -1} = \myvec{0.2 \\ 0.4}
\end{align}
Hence, one of the points of intersection is $\brak{0.2, 0.4}$, and option (B) is correct.

\begin{align}
\frac{dy}{dt} = \lambda\brak{y + 2}
\end{align}

    \begin{multicols}{2}
        \item $\ln 2$
        \item $\ln 3$
        \item $\ln 4$
        \item $\ln 6$
    \end{multicols}
\end{enumerate}

\solution
Rearranging the differential equation into standard linear form:
\begin{align}
\frac{dy}{dt} - \lambda y = 2\lambda
\end{align}

Taking the unilateral Laplace transform on both sides using \eqref{eq:laplace_first_deriv} and \eqref{eq:laplace_const}:
\begin{align}
s Y(s) - y\brak{0} - \lambda Y(s) &= \frac{2\lambda}{s}
\end{align}


Substituting the initial condition $y\brak{0} = 1$:
\begin{align}
\brak{s - \lambda}Y(s) - 1 &= \frac{2\lambda}{s} \\
\brak{s - \lambda}Y(s) &= 1 + \frac{2\lambda}{s} = \frac{s + 2\lambda}{s} \\
Y(s) &= \frac{s + 2\lambda}{s\brak{s - \lambda}}
\end{align}

Decomposing into partial fractions:
\begin{align}
\frac{s + 2\lambda}{s\brak{s - \lambda}} = \frac{A}{s} + \frac{B}{s - \lambda} \implies s + 2\lambda = A\brak{s - \lambda} + B s
\end{align}

\item Evaluating the coefficients:
\begin{align}
\text{At } s = 0: \quad 2\lambda &= A\brak{-\lambda} \implies A = -2 \\
\text{At } s = \lambda: \quad 3\lambda &= B\brak{\lambda} \implies B = 3
\end{align}

Substituting $A$ and $B$ back into $Y(s)$ and applying the inverse Laplace transform using \eqref{eq:laplace_const} and \eqref{eq:laplace_exp_inv}:
\begin{align}
Y(s) &= -\frac{2}{s} + \frac{3}{s - \lambda} \\
y\brak{t} &= 3e^{\lambda t} - 2
\end{align}

Applying the boundary condition $y\brak{1} = 4$:
\begin{align}
4 &= 3e^{\lambda\brak{1}} - 2 \\
6 &= 3e^{\lambda} \\
e^{\lambda} &= 2 \implies \lambda = \ln 2
\end{align}

\item The minimum value of the function:
\begin{align}
f\brak{x} = x + \frac{4}{x}
\end{align}
for $x > 0$ is:
\hfill (GATE~BT~2025)
\begin{enumerate}[itemsep=0pt]
    \begin{multicols}{2}
        \item $1$
        \item $2$
        \item $3$
        \item $4$
    \end{multicols}
\end{enumerate}

\solution
To locate the critical points of $f\brak{x}$, we set the first derivative to zero:
\begin{align}
f'\brak{x} &= \diff{}{x}\brak{x + \frac{4}{x}} = 1 - \frac{4}{x^2} = 0 \\
1 &= \frac{4}{x^2} \implies x^2 = 4
\end{align}
Since $x > 0$, the only admissible critical point is:
\begin{align}
x = 2
\end{align}

Checking the second derivative using \eqref{eq:convexity_check}:
\begin{align}
f''\brak{x} = \frac{d^2}{dx^2}\brak{x + \frac{4}{x}} = \frac{8}{x^3}
\end{align}
For all $x \in \brak{0, \infty}$, $f''\brak{x} > 0$. Thus, $f\brak{x}$ is strictly convex, and $x = 2$ corresponds to a unique global minimum.

Evaluating the function at $x = 2$:
\begin{align}
f\brak{2} = 2 + \frac{4}{2} = 4
\end{align}

\noindent{Numerical Simulation (Gradient Descent):} 
Applying the first-order update relation from \eqref{eq:grad_descent_update} with learning rate $\eta = 0.3$:
\begin{align}
x_{k+1} = x_k - 0.3\brak{1 - \frac{4}{x_k^2}}
\end{align}
Starting from the initial guess $x_0 = 5$, the sequence converges to $x^* = 2$, yielding $f\brak{x^*} = 4$.

\item Alternative Method (Completing the Square):

\begin{align}
f\brak{x} &= x + \frac{4}{x} \\
&= \brak{\sqrt{x}}^2 - 2\brak{\sqrt{x}}\brak{\frac{2}{\sqrt{x}}} + \brak{\frac{2}{\sqrt{x}}}^2 + 2\brak{\sqrt{x}}\brak{\frac{2}{\sqrt{x}}} \\
&= \brak{\sqrt{x} - \frac{2}{\sqrt{x}}}^2 + 4
\end{align}

Since the square of any real number is non-negative:
\begin{align}
\brak{\sqrt{x} - \frac{2}{\sqrt{x}}}^2 \ge 0
\end{align}
with equality holding if and only if:
\begin{align}
\sqrt{x} - \frac{2}{\sqrt{x}} = 0 \implies \sqrt{x} = \frac{2}{\sqrt{x}} \implies x = 2
\end{align}

\item The value of $k$, for which the linear equations $2x + 3y = 6$ and $4x + 6y = 3k$ have at least one solution, is \underline{\hspace{1.5cm}} (answer in integer).
\hfill (GATE~BT~2025)

\solution
The given system of linear equations is:
\begin{align}
2x + 3y &= 6 \\
4x + 6y &= 3k
\end{align}
Expressing in matrix form $\vec{A}\vec{x} = \vec{b}$:
\begin{align}
\myvec{2 & 3 \\ 4 & 6} \myvec{x \\ y} = \myvec{6 \\ 3k}
\end{align}
The augmented matrix is formulated as:
\begin{align}
\augvec{2}{1}{\vec{A} & \vec{b}} = \augvec{2}{1}{2 & 3 & 6 \\ 4 & 6 & 3k}
\end{align}

Applying Gaussian elimination using the elementary row operation $R_2 \to R_2 - 2R_1$:
\begin{align}
\augvec{2}{1}{2 & 3 & 6 \\ 4 & 6 & 3k} \xrightarrow{R_2 \to R_2 - 2R_1} \augvec{2}{1}{2 & 3 & 6 \\ 0 & 0 & 3k - 12}
\end{align}
The coefficient matrix $\vec{A}$ reduces to:
\begin{align}
\vec{A} \xrightarrow{R_2 \to R_2 - 2R_1} \myvec{2 & 3 \\ 0 & 0}
\end{align}
Since there is only one non-zero row in echelon form, $\text{rank}\brak{\vec{A}} = 1$.

From the Rouch\'{e}--Capelli theorem in \eqref{eq:consistent_rank}, for the system to have at least one solution, it must be consistent:
\begin{align}
\text{rank}\brak{\augvec{2}{1}{\vec{A} & \vec{b}}} = \text{rank}\brak{\vec{A}} = 1
\end{align}
This requires the second row of the augmented matrix to be entirely zeros:
\begin{align}
3k - 12 &= 0 \\
3k &= 12 \implies k = 4
\end{align}
For $k = 4$, $\text{rank}\brak{\vec{A}} = \text{rank}\brak{\augvec{2}{1}{\vec{A} & \vec{b}}} = 1 < 2$ (number of variables $n = 2$), yielding infinitely many solutions along the coincident line $2x + 3y = 6$.

\item A thermometer measuring body temperature follows a first-order response with a time constant of $40$ seconds. The instrument will reach $95\%$ of its steady-state output at \underline{\hspace{1.5cm}} seconds (round off to the nearest integer).
\hfill (GATE~BT~2025)
\begin{enumerate}[itemsep=0pt]
    \begin{multicols}{2}
        \item $60$
        \item $80$
        \item $120$
        \item $160$
    \end{multicols}
\end{enumerate}

\solution
The dynamic thermal response of the thermometer is modeled by the standard first-order differential equation given in \eqref{eq:first_order_ode_std}:
\begin{align}
\tau \frac{dy}{dt} + y\brak{t} = y_{ss}
\end{align}
which is analogous to the series $RC$ low-pass circuit in \eqref{eq:rc_circuit_ode}, where the measured temperature $y\brak{t}$ maps to the capacitor output voltage $V_{out}\brak{t}$, the steady-state reading $y_{ss}$ corresponds to $V_{dc}$, and the thermal time constant $\tau = 40\text{ s}$ corresponds to the circuit time constant $RC$.

Converting the system to the $s$-domain equivalent circuit from \eqref{eq:rc_s_domain_vout}:
\begin{align}
Y\brak{s} = \frac{y_{ss}}{s\brak{\tau s + 1}} = y_{ss}\sbrak{\frac{1}{s} - \frac{1}{s + \frac{1}{\tau}}}
\end{align}
Taking the inverse Laplace transform directly yields the transient response from \eqref{eq:first_order_step_sol}:
\begin{align}
y\brak{t} = y_{ss}\brak{1 - e^{-\frac{t}{\tau}}}
\end{align}

The instrument reaches $95\%$ of its steady-state reading when $y\brak{t_{95}} = 0.95 y_{ss}$:
\begin{align}
y_{ss}\brak{1 - e^{-\frac{t_{95}}{\tau}}} &= 0.95 y_{ss} \\
1 - e^{-\frac{t_{95}}{\tau}} &= 0.95 \\
e^{-\frac{t_{95}}{\tau}} &= 0.05 = \frac{1}{20}
\end{align}

Taking the natural logarithm on both sides:
\begin{align}
-\frac{t_{95}}{\tau} &= \ln\brak{0.05} = -\ln\brak{20} \\
t_{95} &= \tau \ln\brak{20}
\end{align}
Substituting the given time constant $\tau = 40\text{ s}$:
\begin{align}
t_{95} &= 40 \ln\brak{20} \\
&\approx 40 \times 2.99573 \approx 119.83\text{ s}
\end{align}
Rounding off to the nearest integer gives $120$ seconds.

\item The output $y\brak{t}$ of a first-order process is governed by the following differential equation
\begin{align}
\tau_p \frac{dy}{dt} + y = K_p f\brak{t}
\end{align}
where $\tau_p$ is a non-zero time constant, $K_p$ is the gain and $f\brak{t}$ is the input with $f\brak{0} = 0$.

Assume $y\brak{0} = 0$. The transfer function for this process is (consider $s$ as the independent variable in the Laplace domain)
\hfill (GATE BT 2025)
\begin{enumerate}[itemsep=1ex]
    \begin{multicols}{2}
        \item $\frac{K_p}{\tau_p s + 1}$
        \item $\frac{\tau_p}{K_p s + 1}$
        \item $\frac{\tau_p}{K_p\brak{s + 1}}$
        \item $\frac{K_p}{\tau_p\brak{s + 1}}$
    \end{multicols}
\end{enumerate}

\solution
The given first-order differential equation governing the process is:
\begin{align}
\tau_p \frac{dy\brak{t}}{dt} + y\brak{t} = K_p f\brak{t}
\end{align}

Taking the unilateral Laplace transform on both sides using \eqref{eq:laplace_first_deriv}:
\begin{align}
\tau_p \sbrak{s Y\brak{s} - y\brak{0}} + Y\brak{s} = K_p F\brak{s}
\end{align}

Applying the given initial condition $y\brak{0} = 0$:
\begin{align}
\tau_p s Y\brak{s} + Y\brak{s} &= K_p F\brak{s} \\
Y\brak{s} \brak{\tau_p s + 1} &= K_p F\brak{s}
\end{align}

The transfer function is the ratio of the output $Y\brak{s}$ to the input $F\brak{s}$:
\begin{align}
\frac{Y\brak{s}}{F\brak{s}} = \frac{K_p}{\tau_p s + 1}
\end{align}

\item Let $m$ and $n$ be fixed real numbers. If the function $y\brak{t} = C_1 e^t + C_2 e^{-t}$ is a solution of
\begin{align}
\frac{d^2 y}{dt^2} + m \frac{dy}{dt} + ny = 0
\end{align}
for any constants $C_1$ and $C_2$, then $m + n$ is equal to
\hfill (GATE~BT~2025)
\begin{enumerate}[itemsep=0pt]
    \begin{multicols}{2}
        \item -2
        \item -1
        \item 0
        \item 1
    \end{multicols}
\end{enumerate}

\solution
The given function is:
\begin{align}
y\brak{t} = C_1 e^t + C_2 e^{-t}
\end{align}

Computing the first and second derivatives with respect to $t$:
\begin{align}
\frac{dy}{dt} &= C_1 e^t - C_2 e^{-t} \\
\frac{d^2 y}{dt^2} &= C_1 e^t + C_2 e^{-t}
\end{align}

Substituting $y$, $\frac{dy}{dt}$, and $\frac{d^2 y}{dt^2}$ into the differential equation:
\begin{align}
\frac{d^2 y}{dt^2} + m \frac{dy}{dt} + ny &= 0 \\
\brak{C_1 e^t + C_2 e^{-t}} + m\brak{C_1 e^t - C_2 e^{-t}} + n\brak{C_1 e^t + C_2 e^{-t}} &= 0
\end{align}

Grouping the terms containing $C_1$ and $C_2$:
\begin{align}
C_1 e^t \brak{1 + m + n} + C_2 e^{-t} \brak{1 - m + n} = 0
\end{align}

Using \eqref{eq:linear_identity} and \eqref{eq:linear_coeffs_zero}, since this equation must hold for any constants $C_1$ and $C_2$:
\begin{align}
1 + m + n &= 0 \label{eq:q45_eq1} \\
1 - m + n &= 0 \label{eq:q45_eq2}
\end{align}

Adding \eqref{eq:q45_eq1} and \eqref{eq:q45_eq2}:
\begin{align}
2 + 2n = 0 \implies n = -1
\end{align}

Subtracting \eqref{eq:q45_eq2} from \eqref{eq:q45_eq1}:
\begin{align}
2m = 0 \implies m = 0
\end{align}

Evaluating $m + n$:
\begin{align}
m + n = 0 + \brak{-1} = -1
\end{align}

\lim_{x \to 0^+} f\brak{x} = \lim_{x \to 0^-} f\brak{x} = f\brak{0}

\begin{align}

f'\brak{0^+} = f'\brak{0^-}

For $x < 0$:

Evaluating $a + b$:

\item In a fermentation process, each mole of glucose is converted to biomass $\brak{\text{CH}_{1.8}\text{O}_{0.5}\text{N}_{0.2}}$, with a biomass yield coefficient of $0.4\text{ C-mol/C-mol}$, according to the unbalanced equation given below.

\text{C}_6\text{H}_{12}\text{O}_6 + \text{NH}_3 + \text{O}_2 \rightarrow \text{CH}_{1.8}\text{O}_{0.5}\text{N}_{0.2} + \text{CO}_2 + \text{H}_2\text{O}

\item Let $a_0 = 0$ and define $a_n = \frac{1}{2}\brak{1 + a_{n-1}}$ for all positive integers $n \ge 1$.
The least value of $n$ for which $\abs{1 - a_n} < \frac{1}{2^{10}}$ is \underline{\hspace{2cm}}. 
\hfill (GATE~BT~2025)

\solution
The given recurrence relation is:
\begin{align}
a_n = \frac{1}{2}\brak{1 + a_{n-1}}, \quad n \ge 1, \quad a_0 = 0
\end{align}

Let $x\brak{n} = a_n u\brak{n}$, where $u\brak{n}$ is the discrete unit step function:
\begin{align}
u\brak{n} = \begin{cases}
1, & n \ge 0 \\
0, & n < 0
\end{cases}
\end{align}
Incrementing the recurrence index to $n \ge 0$:
\begin{align}
a_{n+1} - \frac{1}{2} a_n = \frac{1}{2}, \quad n \ge 0
\end{align}
Multiplying by $u\brak{n}$ yields the causal difference equation:
\begin{align}
x\brak{n+1} - \frac{1}{2} x\brak{n} = \frac{1}{2} u\brak{n}, \quad n \ge 0 \label{eq:diff_eq_causal}
\end{align}
(Equivalently, shifting the index by one sample gives $x\brak{n} - \frac{1}{2} x\brak{n-1} = \frac{1}{2} u\brak{n-1}$ for $n \ge 1$).

The unilateral (one-sided) Z-transform of $x\brak{n}$ is defined as:
\begin{align}
X^+\brak{z} = \mathcal{Z}^+\brak{x\brak{n}} = \sum_{n=0}^{\infty} x\brak{n} z^{-n} \label{eq:unilateral_z_def}
\end{align}
Applying the unilateral Z-transform to the advanced sequence $x\brak{n+1}$:
\begin{align}
\mathcal{Z}^+\brak{x\brak{n+1}} &= \sum_{n=0}^{\infty} x\brak{n+1} z^{-n} \\
&= z \sum_{n=0}^{\infty} x\brak{n+1} z^{-\brak{n+1}} \\
&= z \brak{\sum_{k=0}^{\infty} x\brak{k} z^{-k} - x\brak{0}} \\
&= z X^+\brak{z} - z x\brak{0} \label{eq:z_advance_prop}
\end{align}

Taking the unilateral Z-transform on both sides of \eqref{eq:diff_eq_causal}:
\begin{align}
\brak{z X^+\brak{z} - z x\brak{0}} - \frac{1}{2} X^+\brak{z} &= \frac{1}{2} \mathcal{Z}^+\brak{u\brak{n}} \\
\brak{z - \frac{1}{2}} X^+\brak{z} - z x\brak{0} &= \frac{\frac{1}{2}}{1 - z^{-1}}
\end{align}
Since $a_0 = 0$, the initial condition is $x\brak{0} = a_0 u\brak{0} = 0$:
\begin{align}
\brak{z - \frac{1}{2}} X^+\brak{z} = \frac{\frac{1}{2}}{1 - z^{-1}}
\end{align}
Multiplying both sides by $z^{-1}$:
\begin{align}
\brak{1 - \frac{1}{2} z^{-1}} X^+\brak{z} = \frac{\frac{1}{2} z^{-1}}{1 - z^{-1}}
\end{align}
Solving algebraically for $X^+\brak{z}$:
\begin{align}
X^+\brak{z} = \frac{\frac{1}{2} z^{-1}}{\brak{1 - z^{-1}}\brak{1 - \frac{1}{2} z^{-1}}}, \quad \abs{z} > 1
\end{align}

Performing partial fraction decomposition with $w = z^{-1}$:
\begin{align}
\frac{\frac{1}{2} w}{\brak{1 - w}\brak{1 - \frac{1}{2} w}} = \frac{C_1}{1 - w} + \frac{C_2}{1 - \frac{1}{2} w}
\end{align}
where the residues are evaluated as:
\begin{align}
C_1 &= \sbrak{\frac{\frac{1}{2} w}{1 - \frac{1}{2} w}}_{w = 1} = \frac{1/2}{1/2} = 1 \\
C_2 &= \sbrak{\frac{\frac{1}{2} w}{1 - w}}_{w = 2} = \frac{1}{1 - 2} = -1
\end{align}

Substituting $C_1$ and $C_2$ back into $X^+\brak{z}$:
\begin{align}
X^+\brak{z} = \frac{1}{1 - z^{-1}} - \frac{1}{1 - \frac{1}{2} z^{-1}}
\end{align}

Applying the inverse Z-transform using \eqref{eq:z_step_transform} and \eqref{eq:z_inv_exponential}:
\begin{align}
x\brak{n} = \sbrak{1 - \brak{\frac{1}{2}}^n} u\brak{n} = 1 - 2^{-n}, \quad n \ge 0
\end{align}
Thus, for all $n \ge 0$:
\begin{align}
a_n = 1 - 2^{-n}
\end{align}

Applying the stopping threshold condition $\abs{1 - a_n} < \frac{1}{2^{10}}$:
\begin{align}
\abs{1 - \brak{1 - 2^{-n}}} &< \frac{1}{2^{10}} \\
2^{-n} &< 2^{-10} \\
2^n &> 2^{10} \implies n > 10
\end{align}

Since $n$ must be an integer, the minimal integer satisfying the strict inequality is:
\begin{align}
n = 11
\end{align}
